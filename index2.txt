            if (actionLimits.postDate !== today) { actionLimits.postDate = today; actionLimits.postCount = 0; }
            if (actionLimits.likeDate !== today) { actionLimits.likeDate = today; actionLimits.likeCount = 0; }
            if (actionLimits.shareDate !== today) { actionLimits.shareDate = today; actionLimits.shareCount = 0; }
            saveLimits();
        }
        function saveLimits() {
            localStorage.setItem('michimamo_limits', JSON.stringify(actionLimits));
        }

        // 報告の送信（ポイント付与と上限判定）
        function submitReport() {
            checkLimitsReset();
            
            // 1日の上限チェック（5回）
            if (actionLimits.postCount >= 5) {
                showToast('【上限到達】本日の安全ポイント（投稿）は5回までです。明日またお願いします！');
                closeModal('postModal');
                return;
            }

            const category = document.getElementById('reportCategory').value;
            const photo = document.getElementById('photoInput').files[0];

            if (!photo) {
                showToast('現場の写真を撮影してください。');
                return;
            }

            // ピンをマップに追加
            L.marker([tempLatLng.lat, tempLatLng.lng], {icon: icons[category]}).addTo(map)
             .bindPopup(`<b>${category}</b><br>ユーザー報告`);
             
            // ポイント付与と回数カウント更新
            points += 10;
            actionLimits.postCount++;
            saveLimits();
            updatePointsDisplay();

            showToast(`報告ありがとうございます！10 pt獲得しました（本日 ${actionLimits.postCount}/5 回）`);
            closeModal('postModal');
        }

        // ==========================================
        // 5. ガチャシステム（確率制御・アニメーション）
        // ==========================================
        function playGacha() {
            if (points < 100) {
                showToast('ポイントが足りません。100ptから回せます！');
                return;
            }

            // ポイント消費
            points -= 100;
            updatePointsDisplay();

            // モーダル表示＆アニメーション開始（揺れる）
            const modal = document.getElementById('gachaModal');
            const anim = document.getElementById('gachaAnimation');
            const resultText = document.getElementById('gachaResultText');
            const subText = document.getElementById('gachaSubText');
            const closeBtn = document.getElementById('gachaCloseBtn');
            
            modal.style.display = 'flex';
            anim.className = 'gacha-shake'; 
            anim.textContent = '🎁';
            resultText.style.display = 'none';
            subText.style.display = 'none';
            closeBtn.style.display = 'none';

            // ▼ ガチャの確率抽選処理 ▼
            const rand = Math.random() * 100; // 0.0 〜 100.0 の乱数
            let prizeStr = "";
            let prizePt = 0;
            
            // 1等: 0.0001% (10万pt)
            if (rand <= 0.0001) {
                prizeStr = "✨ 1等 ✨";
                prizePt = 100000;
            } 
            // 2等: 0.01% (1万pt) ※累積確率
            else if (rand <= 0.0101) { 
                prizeStr = "🌟 2等 🌟";
                prizePt = 10000;
            } 
            // 3等: 10% (50pt)
            else if (rand <= 10.0101) { 
                prizeStr = "🎉 3等 🎉";
                prizePt = 50;
            } 
            // 4等: 89.9899% (1pt)
            else {
                prizeStr = "4等";
                prizePt = 1;
            }

            // 当選ポイントの付与
            points += prizePt;

            // 演出（2秒後にパカッと割れて結果表示）
            setTimeout(() => {
                anim.className = 'gacha-pop'; 
                anim.textContent = prizeStr === "4等" ? '💊' : '💎'; // ハズレ枠とアタリ枠でアイコンを変える
                
                resultText.textContent = `${prizePt.toLocaleString()} pt`;
                resultText.style.display = 'block';
                
                subText.textContent = `${prizeStr} おめでとうございます！`;
                subText.style.display = 'block';
                
                closeBtn.style.display = 'block';
                updatePointsDisplay();
            }, 2000);
        }
